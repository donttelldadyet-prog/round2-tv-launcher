cmake_minimum_required(VERSION 3.16)
project(uxair_webos LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED sdl2)
pkg_check_modules(TTF REQUIRED SDL2_ttf)
# Round2: webOS 5+ build. Video and audio go through NDL DirectMedia v2 (libAcbAPI is gone on
# webOS 5), and the modern C++ ABI matches webOS 5's own libraries.
pkg_check_modules(NDL REQUIRED NDL_directmedia)
add_library(lg_media STATIC src/media.cpp src/ndl_glue.c src/shell.cpp src/round2_ui.cpp src/input.cpp src/settings.cpp src/ui_navigation.cpp src/usage_stats.cpp)
target_include_directories(lg_media PUBLIC src ${SDL_INCLUDE_DIRS} ${TTF_INCLUDE_DIRS} ${NDL_INCLUDE_DIRS})
target_link_directories(lg_media PUBLIC ${SDL_LIBRARY_DIRS} ${NDL_LIBRARY_DIRS})
target_link_libraries(lg_media PUBLIC ${SDL_LIBRARIES} ${TTF_LIBRARIES} ${NDL_LIBRARIES} pthread)
target_compile_options(lg_media PRIVATE -Wall -Wextra)
option(BUILD_RECEIVER "Build the AirPlay receiver (requires tools/build-deps.sh)" OFF)
option(BUILD_DEVELOPMENT_PACKAGE "Use a separate development app and AirPlay identity" OFF)
if(BUILD_DEVELOPMENT_PACKAGE)
    target_compile_definitions(lg_media PUBLIC UXAIR_DEV_BUILD)
endif()
if(BUILD_RECEIVER)
    set(DEPS "${CMAKE_SOURCE_DIR}/.cache/deps" CACHE PATH "Private ARM dependencies")
    set(UX "${CMAKE_SOURCE_DIR}/vendor/uxplay")
    add_subdirectory(${UX}/lib/llhttp)
    add_subdirectory(${UX}/lib/playfair)
    add_subdirectory(${UX}/lib/mdnsd)
    file(GLOB CORE CONFIGURE_DEPENDS "${UX}/lib/*.c")
    if(BUILD_DEVELOPMENT_PACKAGE)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        execute_process(
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/instrument-uxplay.py" "${UX}/lib" "${CMAKE_BINARY_DIR}/generated"
            RESULT_VARIABLE PROBE_RESULT
        )
        if(NOT PROBE_RESULT EQUAL 0)
            message(FATAL_ERROR "Pinned UxPlay direct-request instrumentation failed")
        endif()
        foreach(PROBE_FILE raop.c http_request.c httpd.c raop_rtp_mirror.c)
            list(REMOVE_ITEM CORE "${UX}/lib/${PROBE_FILE}")
            list(APPEND CORE "${CMAKE_BINARY_DIR}/generated/${PROBE_FILE}")
        endforeach()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
            "${CMAKE_SOURCE_DIR}/tools/instrument-uxplay.py"
            "${UX}/lib/raop.c" "${UX}/lib/http_request.c" "${UX}/lib/httpd.c" "${UX}/lib/raop_rtp_mirror.c")
    endif()
    add_library(airplay STATIC ${CORE})
    if(BUILD_DEVELOPMENT_PACKAGE)
        target_sources(airplay PRIVATE src/direct_request_diag.c)
        target_include_directories(airplay PRIVATE "${CMAKE_SOURCE_DIR}/src")
    endif()
    target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L _REENTRANT _LARGEFILE64_SOURCE _FILE_OFFSET_BITS=64)
    target_include_directories(airplay PUBLIC "${UX}/lib" PRIVATE "${UX}/lib/llhttp" "${UX}/lib/playfair" "${DEPS}/include")
    target_link_libraries(airplay PUBLIC playfair dnssd llhttp "${DEPS}/lib/libplist-2.0.a" "${DEPS}/lib/libcrypto.a" pthread dl m)
    add_executable(uxair src/main.cpp src/receiver.cpp src/audio.cpp src/h264.cpp
        src/hevc_sample.cpp src/hevc_stream.cpp)
    target_include_directories(uxair PRIVATE "${DEPS}/include")
    target_link_libraries(uxair PRIVATE lg_media airplay "${DEPS}/lib/libavcodec.a" "${DEPS}/lib/libswresample.a" "${DEPS}/lib/libavutil.a")
    # LG's player loads its own older OpenSSL/FFmpeg indirectly. Keep private
    # static symbols out of the dynamic namespace to avoid interposition.
    target_link_options(uxair PRIVATE -Wl,--exclude-libs,ALL)
    set_target_properties(uxair PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
